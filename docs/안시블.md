## 설치 

### group_vars/dev.yml
```bash
# Runtime Versions
java_version: "17"
gradle_version: "8.14.4"

# Node
node_major_version: "20"
node_version: "20.20.2"
npm_version: "10.8.2"

# Packages
docker_version: "29.3.1"
git_version: "2.43.0"

```
### roles/common/tasks/main.yml
```bash

---
# System packages
- name: Update apt cache
  apt:
    update_cache: yes

- name: Install base packages
  apt:
    name:
      - curl
      - unzip
      - git
      - openjdk-17-jdk
    state: present

# Node.js Repository
- name: Add NodeSource repository
  shell: |
    curl -fsSL https://deb.nodesource.com/setup_{{ node_major_version }}.x | bash -
  args:
    executable: /bin/bash

# Node.js Install
- name: Install Node.js
  apt:
    name: nodejs
    state: present

# Verify Versions
- name: Check Java version
  command: java -version
  register: java_version_result
  changed_when: false

- name: Check Node version
  command: node -v
  register: node_version_result
  changed_when: false

- name: Check npm version
  command: npm -v
  register: npm_version_result
  changed_when: false

# Gradle Install
- name: Download Gradle
  get_url:
    url: "https://services.gradle.org/distributions/gradle-{{ gradle_version }}-bin.zip"
    dest: /tmp/gradle.zip

- name: Unarchive Gradle
  unarchive:
    src: /tmp/gradle.zip
    dest: /opt/
    remote_src: yes

- name: Create Gradle symlink
  file:
    src: "/opt/gradle-{{ gradle_version }}/bin/gradle"
    dest: /usr/bin/gradle
    state: link

# Verify Gradle
- name: Check Gradle version
  command: gradle -v
  register: gradle_version_result
  changed_when: false

```
